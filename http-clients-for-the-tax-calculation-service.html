    6. HTTP clients for the tax calculation service

    6.1. Introduction

    This chapter reproduces, without any structural changes, the three successive layers of clients built in the TypeScript course, each refining the previous one: a first client, CLI, where all calculations remain on the server side; a second client, CLI, where the calculation moves to the client side ([business] layer); a third client, compiled for the BROWSER; and finally, a complete client, HTML, with its own views. All four rely on the same NestJS server, which was installed in the previous chapter.

    // dao1.dart
// Ports Dao1.ts: the [dao] layer (Data Access Object) encapsulates all the
// details of HTTP communication with the tax calculation server (URL,
// method, session cookie) behind simple business methods
// (initSession, authenticateUser, calculateTax...).
import 'package:dio/dio.dart';

class Dao1 {
  final Dio _dio;
  final String _sessionCookieName;
  String _sessionCookie = '';

  // constructor - receives the Dio instance to use, as the TypeScript
  // script received the axios instance as a parameter
  Dao1(this._dio) : _sessionCookieName = "connect.sid";

  // init session
  Future<dynamic> initSession() {
    return _getRemoteData(
      'main.php',
      method: 'GET',
      queryParameters: {'action': 'init-session', 'type': 'json'},
    );
  }

  Future<dynamic> authenticateUser(String user, String password) {
    return _getRemoteData(
      'main.php',
      method: 'POST',
      queryParameters: {'action': 'authenticate-user'},
      data: {'user': user, 'password': password},
      contentType: Headers.formUrlEncodedContentType,
    );
  }

  // tax calculation
  Future<dynamic> calculateTax(String married, int children, int salary) {
    return _getRemoteData(
      'main.php',
      method: 'POST',
      queryParameters: {'action': 'calculate-tax'},
      data: {'married': married, 'children': children, 'salary': salary},
      contentType: Headers.formUrlEncodedContentType,
    );
  }

  // list of simulations
  Future<dynamic> listSimulations() {
    return _getRemoteData(
      'main.php',
      method: 'GET',
      queryParameters: {'action': 'list-simulations'},
    );
  }

  // delete a simulation
  Future<dynamic> deleteSimulation(int index) {
    return _getRemoteData(
      'main.php',
      method: 'GET',
      queryParameters: {'action': 'delete-simulation', 'number': index},
    );
  }

  Future<dynamic> _getRemoteData(
    String path, {
    required String method,
    Map<String, dynamic>? queryParameters,
    Object? data,
    String? contentType,
  }) async {
    // for the session cookie
    final options = Options(
      method: method,
      contentType: contentType,
      headers: {'Cookie': _sessionCookie},
      validateStatus: (_) => true,
    );
    // executing the HTTP request
    final response = await _dio.request(
      path,
      queryParameters: queryParameters,
      data: data,
      options: options,
    );
    // we retrieve the session cookie if it exists
    final setCookie = response.headers['set-cookie'];
    if (setCookie != null) {
      for (final header in setCookie) {
        final result = RegExp('^($_sessionCookieName.+?);').firstMatch(header);
        if (result != null) {
          _sessionCookie = result.group(1)!;
          break;
        }
      }
    }
    // the server's response is in [response.data]
    return response.data;
  }
}

    // business.dart
// Ports Business.ts: the tax calculation, done entirely on the client side
// (in the browser in TypeScript, here in the Dart program), from
// the raw tax data obtained only once from the server via
// [Dao2.getAdminData] (see dao2.dart).
class Business {
  // [dynamic]: the exact structure of the data sent by the tax
  // server (rates, ceilings...) is not documented here; as in
  // TypeScript, we deliberately stay flexible on this type
  final Map<String, dynamic> taxAdminData;

  Business(this.taxAdminData);

  // tax calculation
  // --------------------------------------------------------------------------
  // married: yes, no
  // children: number of children
  // salary: annual salary
  Map<String, dynamic> calculateTax(String married, int children, num salary) {
    // tax calculation with children
    final result1 = _calculateTax2(married, children, salary);
    final tax1 = result1['tax'] as num;
    // tax calculation without the children
    Map<String, dynamic> result2;
    num tax2;
    num halfShareCeiling;
    if (children != 0) {
      result2 = _calculateTax2(married, 0, salary);
      tax2 = result2['tax'] as num;
      // applying the family quotient cap
      halfShareCeiling = taxAdminData['halfShareQfCeiling'] as num;
      if (children < 3) {
        // halfShareCeiling for the first 2 children
        tax2 = tax2 - children * halfShareCeiling;
      } else {
        // halfShareCeiling for the first 2 children, double for the following ones
        tax2 = tax2 - 2 * halfShareCeiling - (children - 2) * 2 * halfShareCeiling;
      }
    } else {
      // no recalculation of the tax
      tax2 = tax1;
      result2 = result1;
    }
    // we take the higher tax in [tax1, tax2]
    num tax;
    num rate;
    num surcharge;
    if (tax1 > tax2) {
      tax = tax1;
      rate = result1['rate'] as num;
      surcharge = result1['surcharge'] as num;
    } else {
      surcharge = tax2 - tax1 + (result2['surcharge'] as num);
      tax = tax2;
      rate = result2['rate'] as num;
    }
    // calculation of a possible discount
    final discount = _getDiscount(married, tax);
    tax -= discount;
    // calculation of a possible tax reduction
    final reduction = _getReduction(married, salary, children, tax);
    tax -= reduction;
    // result
    return {
      'tax': tax.floor(),
      'surcharge': surcharge,
      'discount': discount,
      'reduction': reduction,
      'rate': rate,
    };
  }

  // --------------------------------------------------------------------------
  Map<String, dynamic> _calculateTax2(String married, int children, num salary) {
    // number of shares
    married = married.toLowerCase();
    num shares;
    if (married == "yes") {
      shares = children / 2 + 2;
    } else {
      shares = children / 2 + 1;
    }
    // 1 share per child from the 3rd one
    if (children >= 3) {
      // one more half-share for each child from the 3rd one
      shares += 0.5 * (children - 2);
    }
    // taxable income
    final taxableIncome = _getTaxableIncome(salary);
    // surcharge
    num surcharge = (taxableIncome - 0.9 * salary).floor();
    // for rounding issues
    if (surcharge < 0) {
      surcharge = 0;
    }
    // family quotient
    final quotient = taxableIncome / shares;
    // tax calculation
    final List<num> limits = List<num>.from(taxAdminData['limits'] as List);
    final List<num> coeffR = List<num>.from(taxAdminData['coeffR'] as List);
    final List<num> coeffN = List<num>.from(taxAdminData['coeffN'] as List);
    // is put at the end of the [limits] list to stop the following loop
    limits[limits.length - 1] = quotient;
    // looking for the tax rate
    var i = 0;
    while (quotient > limits[i]) {
      i++;
    }
    // because we placed quotient at the end of limits, the previous loop
    // cannot go beyond the limits list
    // now we can calculate the tax
    final tax = (taxableIncome * coeffR[i] - shares * coeffN[i]).floor();
    // result
    return {'tax': tax, 'surcharge': surcharge, 'rate': coeffR[i]};
  }

  // taxableIncome = annualSalary - allowance (the allowance has a min and a max)
  num _getTaxableIncome(num salary) {
    // allowance of 10% of the salary
    num allowance = 0.1 * salary;
    // this allowance cannot exceed tenPercentAllowanceMax
    final allowanceMax = taxAdminData['tenPercentAllowanceMax'] as num;
    if (allowance > allowanceMax) {
      allowance = allowanceMax;
    }
    // the allowance cannot be lower than tenPercentAllowanceMin
    final allowanceMin = taxAdminData['tenPercentAllowanceMin'] as num;
    if (allowance < allowanceMin) {
      allowance = allowanceMin;
    }
    // taxable income
    final taxableIncome = salary - allowance;
    return taxableIncome.floor();
  }

  // calculates a possible discount
  num _getDiscount(String married, num taxes) {
    // initially, a zero discount
    num discount = 0;
    // maximum tax amount to get the discount
    final taxCeilingForDiscount = married == "yes"
        ? taxAdminData['coupleTaxCeilingForDiscount'] as num
        : taxAdminData['singleTaxCeilingForDiscount'] as num;
    if (taxes < taxCeilingForDiscount) {
      // maximum amount of the discount
      final discountCeiling = married == "yes"
          ? taxAdminData['coupleDiscountCeiling'] as num
          : taxAdminData['singleDiscountCeiling'] as num;
      // theoretical discount
      discount = discountCeiling - 0.75 * taxes;
      // the discount cannot exceed the amount of the tax
      if (discount > taxes) {
        discount = taxes;
      }
      // no discount < 0
      if (discount < 0) {
        discount = 0;
      }
    }
    // result - ceil() is the Dart equivalent of Math.ceil()
    return discount.ceil();
  }

  // calculates a possible reduction
  num _getReduction(String married, num salary, int children, num taxes) {
    // the income ceiling to be entitled to the 20% reduction
    num incomeCeilingForReduction = married == "yes"
        ? taxAdminData['coupleIncomeCeilingForReduction'] as num
        : taxAdminData['singleIncomeCeilingForReduction'] as num;
    final halfShareReductionValue = taxAdminData['halfShareReductionValue'] as num;
    incomeCeilingForReduction += children * halfShareReductionValue;
    if (children > 2) {
      incomeCeilingForReduction += (children - 2) * halfShareReductionValue;
    }
    // taxable income
    final taxableIncome = _getTaxableIncome(salary);
    // reduction
    num reduction = 0;
    if (taxableIncome < incomeCeilingForReduction) {
      // 20% reduction
      reduction = 0.2 * taxes;
    }
    return reduction.ceil();
  }
}

    Let’s break down this code:

    • line 12: [Business(this.taxAdminData)]: the constructor receives the raw tax data retrieved by DAO only once—no further network calls are required afterward to calculate a tax.
    • line 19: [Map<String, dynamic> calculateTax(String married, int children, num salary)]: a faithful, line-by-line port of the TypeScript algorithm—same steps ([_calculateTax2], [_getTaxableIncome], [_getDiscount], [_getReduction]), same logic for capping the family quotient.
    • [.floor()]/[.ceil()] (methods of [num]) replace [Math.floor(...)]/[Math.ceil(...)]: In Dart, these are instance methods on the number itself, not static methods of a utility class.
    • The KEYS of the returned [Map] ([tax], [surcharge], [discount], [reduction]...) have the same plain ASCII names as the Dart IDENTIFIERS (variable and parameter names)—a string may contain accents in Dart, but an identifier may not (see simulation.dart, later in this chapter).

// authenticateUser, but exposes [getAdminData] instead of
// [calculateTax]/[listSimulations]/[deleteSimulation] - the calculation and

import 'package:dio/dio.dart';

class Dao2 {
  final Dio _dio;
  final String _sessionCookieName;
  String _sessionCookie = '';

  Dao2(this._dio) : _sessionCookieName = "connect.sid";

  Future<dynamic> initSession() {
    return _getRemoteData(
      'main.php',
      method: 'GET',
      queryParameters: {'action': 'init-session', 'type': 'json'},
    );
  }

  Future<dynamic> authenticateUser(String user, String password) {
    return _getRemoteData(
      'main.php',
      method: 'POST',
      queryParameters: {'action': 'authenticate-user'},
      data: {'user': user, 'password': password},
      contentType: Headers.formUrlEncodedContentType,
    );
  }

  Future<dynamic> getAdminData() {
    return _getRemoteData(
      'main.php',
      method: 'GET',
      queryParameters: {'action': 'get-admindata'},
    );
  }

  Future<dynamic> _getRemoteData(
    String path, {
    required String method,
    Map<String, dynamic>? queryParameters,
    Object? data,
    String? contentType,
  }) async {
    final options = Options(
      method: method,
      contentType: contentType,
      headers: {'Cookie': _sessionCookie},
      validateStatus: (_) => true,
    );
    final response = await _dio.request(
      path,
      queryParameters: queryParameters,
      data: data,

// Ports tax client/http client 3/src/Dao3.ts.
//
// Difference with dao2.dart: this file is meant to be compiled for the
// BROWSER (with `dart compile js`, see the README of this folder), not
// run on the CLI with `dart run`. In a browser, the SESSION COOKIE
// is handled automatically by the browser itself (as for any
// web page) as soon as sending them is allowed - so there is no longer
// any need for the MANUAL cookie tracking done by hand in dao1.dart/dao2.dart
// (exactly the reasoning, already present, that allowed Dao3.ts to
// remove all the cookie-handling code of Dao2.ts).
//
// [mapping with TypeScript] "axios.defaults.withCredentials = true"
// (set in main3.dart, not here) corresponds, on the Dart [http] package side, to
// the [withCredentials] option of the web-specific client,
// [BrowserClient] (package:http/browser_client.dart) - a class that
// exists ONLY when the code is compiled for the web, unlike
// http.Client, which works everywhere.
import 'dart:convert';
import 'package:http/http.dart' as http;

class Dao3 {
  final http.Client _client;
  final String _baseUrl;

  Dao3(this._client, this._baseUrl);

  Future<dynamic> initSession() {
    return _getRemoteData(
      'main.php',
      method: 'GET',
      queryParameters: {'action': 'init-session', 'type': 'json'},
    );
  }

  Future<dynamic> authenticateUser(String user, String password) {
    return _getRemoteData(
      'main.php',
      method: 'POST',
      queryParameters: {'action': 'authenticate-user'},
      body: {'user': user, 'password': password},
    );
  }

  Future<dynamic> getAdminData() {
    return _getRemoteData(
      'main.php',
      method: 'GET',
      queryParameters: {'action': 'get-admindata'},
    );
  }

  Future<dynamic> _getRemoteData(
    String path, {
    required String method,
    Map<String, dynamic>? queryParameters,
    Map<String, String>? body,
  }) async {
    final uri = Uri.parse('$_baseUrl/$path').replace(
      queryParameters: queryParameters?.map((k, v) => MapEntry(k, '$v')),
    );
    final http.Response response;
    if (method == 'POST') {
      response = await _client.post(uri, body: body);
    } else {
      response = await _client.get(uri);
    }
    // [difference with the CLI client] no manual reading of the set-cookie
    // header here: in a browser, response.headers does not even expose it
    // (header protected by the browser, invisible to the
    // JavaScript/Dart code, precisely because it is already handled by itself)
    return jsonDecode(response.body);
  }
}

    // app_html.dart
// Ports AppHtml.ts: the controller of the HTML application. It orchestrates the
// calls to the server ([dao] layer, dao3.dart), the calculations
// ([business] layer, business.dart) and the display of the views ([views] layer,
// views.dart).
//
// Like main3.dart, this client calls the NestJS server for only three
// actions: [init-session], [authenticate-user] and [get-admindata].
// Once the tax data (rates, ceilings...) has been retrieved with
// [get-admindata], it is the [business] layer that calculates the tax in the
// browser, with no new network call. The list of simulations is also
// kept only in the browser (field [simulations]),
// exactly as main3.dart does with its local variable
// [simulations] (`simulations.add(...)`, `simulations.removeAt(...)`).
import 'dart:js_interop';
import 'package:http/browser_client.dart';
import 'dao3.dart';
import 'business.dart';
import 'simulation.dart';
import 'views.dart';
import 'package:web/web.dart' as web;

class AppHtml {
  final Dao3 dao;
  final Views views;
  // [business] layer: only exists once the tax data has been retrieved
  // (after a successful authentication) - see [_initBusiness]. A
  // NULLABLE field (Business?), not a late field ([late]): unlike
  // Views.ts where [business?: Business] simply stays "undefined" as long as it
  // is not assigned, a Dart [late] field would throw an error if it were
  // READ before assignment (which is precisely what [_endSession] does when
  // resetting it) - so the nullable type is the right choice here, not
  // just a matter of style.
  Business? _business;
  // list of the simulations already calculated, kept in the browser
  List<Simulation> _simulations = [];

  // [difference with TypeScript] AppHtml.ts configures a GLOBAL axios
  // instance (axios.defaults.timeout/baseURL/withCredentials) then hands it
  // to the Dao. Here, [BrowserClient] (package:http/browser_client.dart, already
  // met in main3.dart) carries the equivalent of
  // "axios.defaults.withCredentials = true"; there is no direct
  // equivalent of a "global timeout" on http.Client - it would be applied case
  // by case with `.timeout(Duration(...))` on each call, as in the
  // [http] chapter (dart_fetch_01.dart).
  AppHtml()
      : dao = Dao3(
          BrowserClient()..withCredentials = true,
          'http://localhost:3000',
        ),
        views = Views(web.document.getElementById('app') as web.HTMLElement);

  // entry point of the application
  Future<void> start() async {
    try {
      // init session - mandatory before any other action (see main.php)
      final response = await dao.initSession() as Map<String, dynamic>;
      if (response['status'] != 700) {
        views.authenticationView(
          _formatError(response['response']),
          (onSubmit: (user, password) => _authenticate(user, password)),
        );
        return;
      }
    } catch (error) {
      views.authenticationView(
        "Unable to contact the server ($error)",
        (onSubmit: (user, password) => _authenticate(user, password)),
      );
      return;
    }
    // session initialized: we display the authentication
    _showAuthentication();
  }

  // --------------------------------------------------------------------
  // [authentication] view
  // --------------------------------------------------------------------
  void _showAuthentication([String? error]) {
    views.authenticationView(
      error,
      (onSubmit: (user, password) => _authenticate(user, password)),
    );
  }

  Future<void> _authenticate(String user, String password) async {
    try {
      final response = await dao.authenticateUser(user, password)
          as Map<String, dynamic>;
      if (response['status'] == 200) {
        // successful authentication: as in main3.dart, we now retrieve,
        // once and for all, the tax data
        // needed by the [business] layer
        await _initBusiness();
      } else {
        // failure (wrong login/password, or missing parameters)
        _showAuthentication(_formatError(response['response']));
      }
    } catch (error) {
      _showAuthentication("Unable to contact the server ($error)");
    }
  }

  // retrieves the tax administration data and instantiates the
  // [business] layer with this data - a single server call for the whole
  // session, exactly like main3.dart
  Future<void> _initBusiness() async {
    try {
      final response = await dao.getAdminData() as Map<String, dynamic>;
      if (response['status'] == 1000) {
        _business = Business(response['response'] as Map<String, dynamic>);
        // new work session: we start again from an empty list
        _simulations = [];
        _showCalculation();
      } else {
        _showAuthentication(_formatError(response['response']));
      }
    } catch (error) {
      _showAuthentication("Unable to contact the server ($error)");
    }
  }

  // --------------------------------------------------------------------
  // [tax calculation] view
  // --------------------------------------------------------------------
  void _showCalculation([Simulation? result, String? error]) {
    views.calculationView(
      result,
      error,
      (
        onSubmit: (married, children, salary) =>
            _calculate(married, children, salary),
        onList: () => _showList(),
        onEndSession: () => _endSession(),
      ),
    );
  }

  // tax calculation done entirely in the browser ([business]
  // layer): no call to the server, unlike the first version
  // of this client (http client 1 / http client 2)
  void _calculate(String married, int children, int salary) {
    // _business has necessarily been initialized: we only reach the [calculation] view
    // after _initBusiness (so the [!] is safe here, like the TypeScript
    // [!] on [this.business!])
    final result = _business!.calculateTax(married, children, salary);
    // [difference with TypeScript] Views.ts builds the simulation with a
    // simple object spread: `{ married, children, salary, ...result }`.
    // Here, [Simulation] is a RECORD (simulation.dart), not a Map: a
    // record cannot be "completed" by spreading a Map, and
    // its fields are typed (num, int...) whereas
    // [result], returned by business.dart, is an untyped Map whose keys
    // merely HAVE the same names ('tax', 'surcharge'..., see business.dart). So we
    // rebuild the record field by field - a little more verbose
    // than a spread, but entirely checked by the compiler (a forgotten
    // or wrongly typed field is a compilation error, not a surprise at
    // runtime).
    final Simulation simulation = (
      married: married,
      children: children,
      salary: salary,
      tax: result['tax'] as num,
      surcharge: result['surcharge'] as num,
      discount: result['discount'] as num,
      reduction: result['reduction'] as num,
      rate: result['rate'] as num,
    );
    // storage in the browser (like `simulations.add(...)` in
    // main3.dart)
    _simulations.add(simulation);
    _showCalculation(simulation);
  }

  // --------------------------------------------------------------------
  // [list of simulations] view
  // --------------------------------------------------------------------
  // no more server calls here: the list is already entirely in
  // [_simulations]
  void _showList([String? error]) {
    views.listView(
      _simulations,
      error,
      (
        onDelete: (number) => _deleteSimulation(number),
        onCalculation: () => _showCalculation(),
        onEndSession: () => _endSession(),
      ),
    );
  }

  void _deleteSimulation(int number) {
    // local deletion from [_simulations], like
    // `simulations.removeAt(number)` in main3.dart - still no
    // server call
    _simulations.removeAt(number);
    _showList();
  }

  // --------------------------------------------------------------------
  // end of session
  // --------------------------------------------------------------------
  void _endSession() {
    // end of session on the client side only: we forget the [business] layer
    // (and therefore the tax data) and the simulations already calculated, then
    // we go back to the authentication screen. This client does not need to
    // call the server for this.
    _business = null;
    _simulations = [];
    _showAuthentication();
  }

  // --------------------------------------------------------------------
  // formats the [response] field of an error response from the server, which
  // is either a string or a list of strings (see PHP controllers)
  // --------------------------------------------------------------------
  String _formatError(Object? response) {
    if (response is List) {
      return response.join(' ; ');
    }
    return '$response';
  }
}

// passed to addEventListener must be converted into a JavaScript function with

// the same rule.
void main() {
  web.window.addEventListener(
    'DOMContentLoaded',
    ((web.Event event) {
      AppHtml().start();
    }).toJS,
  );
}

    6.6. Conclusion

    The four clients in this chapter illustrate, with Dart, exactly the same progression as with TypeScript: from server-side processing to client-side processing, then from the CLI script to the full web client. The key differences to note are the manual tracking of the session cookie (necessary in CLI, unnecessary in a browser), the [.toJS] conversion enforced by [dart:js_interop] to bridge the Dart/JavaScript boundary in views, and the complete absence of polyfills during web compilation—a clear advantage of the Dart toolchain over the webpack/babel pipeline.
