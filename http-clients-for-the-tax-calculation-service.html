    6. HTTP clients for the tax calculation service

    6.1. Introduction

    This chapter reproduces, without any structural changes, the three successive layers of clients built in the TypeScript course, each refining the previous one: a first client, CLI, where all calculations remain on the server side; a second client, CLI, where the calculation moves to the client side ([métier] layer); a third client, compiled for NAVIGATEUR; and finally, a complete client, HTML, with its own views. All four rely on the same NestJS server, which was installed in the previous chapter.

    // dao1.dart
// Port Dao1.ts: The [dao] layer (Data Access Object) encapsulates all
// communication details (HTTP) with the tax calculation server (URL,
// method, session cookie) behind simple business methods
// (initSession, authentifierUtilisateur, calculerImpot...).
import 'package:dio/dio.dart';

class Dao1 {
  final Dio _dio;
  final String _sessionCookieName;
  String _sessionCookie = '';

  // constructor—receives the Dio instance to be used, just like the script
  // TypeScript received the axios instance as a parameter
  Dao1(this._dio) : _sessionCookieName = "connect.sid";

  // init session
  Future<dynamic> initSession() {
    return _getRemoteData(
      'main.php',
      method: 'GET',
      queryParameters: {'action': 'init-session', 'type': 'json'},
    );
  }

  Future<dynamic> authentifierUtilisateur(String user, String password) {
    return _getRemoteData(
      'main.php',
      method: 'POST',
      queryParameters: {'action': 'authentifier-utilisateur'},
      data: {'user': user, 'password': password},
      contentType: Headers.formUrlEncodedContentType,
    );
  }

  // tax calculation
  Future<dynamic> calculerImpot(String marie, int enfants, int salaire) {
    return _getRemoteData(
      'main.php',
      method: 'POST',
      queryParameters: {'action': 'calculer-impot'},
      data: {'marié': marie, 'enfants': enfants, 'salaire': salaire},
      contentType: Headers.formUrlEncodedContentType,
    );
  }

  // list of simulations
  Future<dynamic> listeSimulations() {
    return _getRemoteData(
      'main.php',
      method: 'GET',
      queryParameters: {'action': 'lister-simulations'},
    );
  }

  // delete a simulation
  Future<dynamic> supprimerSimulation(int index) {
    return _getRemoteData(
      'main.php',
      method: 'GET',
      queryParameters: {'action': 'supprimer-simulation', 'numéro': index},
    );
  }

  Future<dynamic> _getRemoteData(
    String path, {
    required String method,
    Map<String, dynamic>? queryParameters,
    Object? data,
    String? contentType,
  }) async {
    // for the session cookie
    final options = Options(
      method: method,
      contentType: contentType,
      headers: {'Cookie': _sessionCookie},
      validateStatus: (_) => true,
    );
    // execute the HTTP request
    final response = await _dio.request(
      path,
      queryParameters: queryParameters,
      data: data,
      options: options,
    );
    // retrieve the session cookie if it exists
    final setCookie = response.headers['set-cookie'];
    if (setCookie != null) {
      for (final entete in setCookie) {
        final resultat = RegExp('^($_sessionCookieName.+?);').firstMatch(entete);
        if (resultat != null) {
          _sessionCookie = resultat.group(1)!;
          break;
        }
      }
    }
    // the server's response is in [response.data]
    return response.data;
  }
}

    // metier.dart
// Port Metier.ts: tax calculation, performed entirely on the client side
// (in the browser as TypeScript, here in the Dart program), based on
// the raw tax data retrieved once from the server via
// [Dao2.getAdminData] (see dao2.dart).
class Metier {
  // [dynamic]: The exact structure of the tax data sent by the server
  // tax-related data (rates, thresholds, etc.) is not documented here; as in
  // TypeScript, we are deliberately keeping things flexible regarding this type
  final Map<String, dynamic> taxAdminData;

  Metier(this.taxAdminData);

  // tax calculation
  // --------------------------------------------------------------------------
  // married: yes, no
  // children: number of children
  // salary: annual salary
  Map<String, dynamic> calculerImpot(String marie, int enfants, num salaire) {
    // tax calculation with children
    final result1 = _calculerImpot2(marie, enfants, salaire);
    final impot1 = result1['impôt'] as num;
    // Tax calculation without children
    Map<String, dynamic> result2;
    num impot2;
    num plafondDemiPart;
    if (enfants != 0) {
      result2 = _calculerImpot2(marie, 0, salaire);
      impot2 = result2['impôt'] as num;
      // Application of the family quotient cap
      plafondDemiPart = taxAdminData['plafondQfDemiPart'] as num;
      if (enfants < 3) {
        // plafondDemiPart for the first 2 children
        impot2 = impot2 - enfants * plafondDemiPart;
      } else {
        // plafondDemiPart for the first 2 children, double that amount for subsequent children
        impot2 = impot2 - 2 * plafondDemiPart - (enfants - 2) * 2 * plafondDemiPart;
      }
    } else {
      // no tax recalculation
      impot2 = impot1;
      result2 = result1;
    }
    // the highest tax amount is used from [impot1, impot2]
    num impot;
    num taux;
    num surcote;
    if (impot1 > impot2) {
      impot = impot1;
      taux = result1['taux'] as num;
      surcote = result1['surcôte'] as num;
    } else {
      surcote = impot2 - impot1 + (result2['surcôte'] as num);
      impot = impot2;
      taux = result2['taux'] as num;
    }
    // calculation of a possible tax deduction
    final decote = _getDecote(marie, impot);
    impot -= decote;
    // Calculation of a potential tax reduction
    final reduction = _getReduction(marie, salaire, enfants, impot);
    impot -= reduction;
    // Result
    return {
      'impôt': impot.floor(),
      'surcôte': surcote,
      'décôte': decote,
      'réduction': reduction,
      'taux': taux,
    };
  }

  // --------------------------------------------------------------------------
  Map<String, dynamic> _calculerImpot2(String marie, int enfants, num salaire) {
    // number of shares
    marie = marie.toLowerCase();
    num nbParts;
    if (marie == "oui") {
      nbParts = enfants / 2 + 2;
    } else {
      nbParts = enfants / 2 + 1;
    }
    // 1 share per child starting with the third
    if (enfants >= 3) {
      // half an additional share for each child starting with the third
      nbParts += 0.5 * (enfants - 2);
    }
    // taxable income
    final revenuImposable = _getRevenuImposable(salaire);
    // surcharge
    num surcote = (revenuImposable - 0.9 * salaire).floor();
    // for rounding issues
    if (surcote < 0) {
      surcote = 0;
    }
    // family quotient
    final quotient = revenuImposable / nbParts;
    // tax calculation
    final List<num> limites = List<num>.from(taxAdminData['limites'] as List);
    final List<num> coeffR = List<num>.from(taxAdminData['coeffR'] as List);
    final List<num> coeffN = List<num>.from(taxAdminData['coeffN'] as List);
    //is placed at the end of the list [limites] to stop the following loop
    limites[limites.length - 1] = quotient;
    // lookup of tax rate
    var i = 0;
    while (quotient > limites[i]) {
      i++;
    }
    // Since the quotient was placed at the end of the limits, the loop
    // cannot overflow the limits list
    // Now we can calculate the tax
    final impot = (revenuImposable * coeffR[i] - nbParts * coeffN[i]).floor();
    // Result
    return {'impôt': impot, 'surcôte': surcote, 'taux': coeffR[i]};
  }

  // revenuImposable = salaireAnnuel - tax deduction (the tax deduction has a minimum and a maximum)
  num _getRevenuImposable(num salaire) {
    // 10% pay cut
    num abattement = 0.1 * salaire;
    // this deduction cannot exceed abattementDixPourCentMax
    final abattementMax = taxAdminData['abattementDixPourCentMax'] as num;
    if (abattement > abattementMax) {
      abattement = abattementMax;
    }
    // The deduction cannot be less than abattementDixPourcentMin
    final abattementMin = taxAdminData['abattementDixPourcentMin'] as num;
    if (abattement < abattementMin) {
      abattement = abattementMin;
    }
    // taxable income
    final revenuImposable = salaire - abattement;
    return revenuImposable.floor();
  }

  // calculates any discount
  num _getDecote(String marie, num impots) {
    // Initially, a zero discount
    num decote = 0;
    // maximum tax amount to qualify for the discount
    final plafondImpotPourDecote = marie == "oui"
        ? taxAdminData['plafondImpotCouplePourDecote'] as num
        : taxAdminData['plafondImpotCelibatairePourDecote'] as num;
    if (impots < plafondImpotPourDecote) {
      // maximum discount amount
      final plafondDecote = marie == "oui"
          ? taxAdminData['plafondDecoteCouple'] as num
          : taxAdminData['plafondDecoteCelibataire'] as num;
      // theoretical discount
      decote = plafondDecote - 0.75 * impots;
      // The discount cannot exceed the tax amount
      if (decote > impots) {
        decote = impots;
      }
      // no discount < 0
      if (decote < 0) {
        decote = 0;
      }
    }
    // Result: `ceil()` is the Dart equivalent of `Math.ceil()`
    return decote.ceil();
  }

  // Calculates a potential reduction
  num _getReduction(String marie, num salaire, int enfants, num impots) {
    // the income threshold for eligibility for the 20% reduction
    num plafondRevenuPourReduction = marie == "oui"
        ? taxAdminData['plafondRevenusCouplePourReduction'] as num
        : taxAdminData['plafondRevenusCelibatairePourReduction'] as num;
    final valeurReducDemiPart = taxAdminData['valeurReducDemiPart'] as num;
    plafondRevenuPourReduction += enfants * valeurReducDemiPart;
    if (enfants > 2) {
      plafondRevenuPourReduction += (enfants - 2) * valeurReducDemiPart;
    }
    // taxable income
    final revenuImposable = _getRevenuImposable(salaire);
    // reduction
    num reduction = 0;
    if (revenuImposable < plafondRevenuPourReduction) {
      // 20% reduction
      reduction = 0.2 * impots;
    }
    return reduction.ceil();
  }
}

    Let’s break down this code:

    • line 12: [Metier(this.taxAdminData)]: the constructor receives the raw tax data retrieved by DAO only once—no further network calls are required afterward to calculate a tax.
    • line 19: [Map<String, dynamic> calculerImpot(String marie, int enfants, num salaire)]: a faithful, line-by-line port of the TypeScript algorithm—same steps ([_calculerImpot2], [_getRevenuImposable], [_getDecote], [_getReduction]), same logic for capping the family quotient.
    • [.floor()]/[.ceil()] (methods of [num]) replace [Math.floor(...)]/[Math.ceil(...)]: In Dart, these are instance methods on the number itself, not static methods of a utility class.
    • The CLÉS instances of the rendered [Map] ([impot], [surcote]...) remain accented ([impôt], [surcôte], [décôte], [réduction]): only the IDENTIFIANTS Dart (variable and parameter names) are unaccented—a string can still contain accents in Dart (see simulation.dart, later in this chapter).

// Port AppHtml.ts: the HTML application controller. It orchestrates the
// server calls ([dao] layer, dao3.dart layer), calculations (
// [métier], metier.dart) and the display of views (layer [vues],
// vues.dart).
//
// Like main3.dart, this client only makes requests to the server NestJS for three
// actions: [init-session], [authentifier-utilisateur], and [get-admindata].
// Once the tax data (rates, thresholds, etc.) has been retrieved using
// [get-admindata], the [métier] layer calculates the tax in the
// browser, without making another network request. Is the list of simulations
// also stored solely in the browser (field [simulations]),
// exactly as main3.dart does with its local variable
// [simulations] (`simulations.add(...)`, `simulations.removeAt(...)`).
import 'dart:js_interop';
import 'package:http/browser_client.dart';
import 'dao3.dart';
import 'metier.dart';
import 'simulation.dart';
import 'vues.dart';
import 'package:web/web.dart' as web;

class AppHtml {
  final Dao3 dao;
  final Vues vues;
  // [métier] layer: exists only after tax data has been retrieved
  // (after successful authentication) — see [initialiserMetier]. A
  // field NULLABLE (Occupation?), not a late field ([late]): unlike
  // to Vues.ts, where [métier?: Métier] simply remains "undefined" as long as
  // is not assigned, a Dart field named [late] would throw an error if it were
  // LISAIT before assignment (which is exactly what [finSession] does by
  // and thereby nulling it)—so the nullable type is the right choice here, not
  // just a matter of style.
  Metier? _metier;
  // list of simulations already calculated, stored in the browser
  List<Simulation> _simulations = [];

  // [différence avec TypeScript] AppHtml.ts configures an instance GLOBALE
  // from Axios (axios.defaults.timeout/baseURL/withCredentials) and then passes it
  // to the Dao. Here, [BrowserClient] (package:http/browser_client.dart, previously
  // encountered in main3.dart) carries the equivalent of
  // "axios.defaults.withCredentials = true"; there is no equivalent
  // that directly corresponds to a “global timeout” on http.Client—it would be applied on a case-by-case basis
  // on a case-by-case basis with `.timeout(Duration(...))` on each call, as in the
  // chapter [http] (dart_fetch_01.dart).
  AppHtml()
      : dao = Dao3(
          BrowserClient()..withCredentials = true,
          'http://localhost:3000',
        ),
        vues = Vues(web.document.getElementById('app') as web.HTMLElement);

  // application entry point
  Future<void> demarrer() async {
    try {
      // init session—required before any other action (see main.php)
      final reponse = await dao.initSession() as Map<String, dynamic>;
      if (reponse['état'] != 700) {
        vues.vueAuthentification(
          _formaterErreur(reponse['réponse']),
          (onValider: (user, password) => _authentifier(user, password)),
        );
        return;
      }
    } catch (erreur) {
      vues.vueAuthentification(
        "Impossible de contacter le serveur ($erreur)",
        (onValider: (user, password) => _authentifier(user, password)),
      );
      return;
    }
    // session initialized: display authentication
    _afficherAuthentification();
  }

  // view [authentification]
  // --------------------------------------------------------------------
  void _afficherAuthentification([String? erreur]) {
    vues.vueAuthentification(
      erreur,
      (onValider: (user, password) => _authentifier(user, password)),
    );

  Future<void> _authentifier(String user, String password) async {
    try {
      final reponse = await dao.authentifierUtilisateur(user, password)
          as Map<String, dynamic>;
      if (reponse['état'] == 200) {
        // Authentication successful: as in main3.dart, we retrieve
        // now, once and for all, the tax data
        // required by the layer [métier]
        await _initialiserMetier();
      } else {
        // failure (incorrect username/password, or missing parameters)
        _afficherAuthentification(_formaterErreur(reponse['réponse']));
      }
    } catch (erreur) {
      _afficherAuthentification("Impossible de contacter le serveur ($erreur)");
    }
  }

  // [métier] layer with this data—a single server call for the entire
  // session, exactly like main3.dart

    try {
      final reponse = await dao.getAdminData() as Map<String, dynamic>;
      if (reponse['état'] == 1000) {
        _metier = Metier(reponse['réponse'] as Map<String, dynamic>);
        // new work session: we start with an empty list

  // view [calcul de l'impôt]
  // --------------------------------------------------------------------
  void _afficherCalcul([Simulation? resultat, String? erreur]) {
    vues.vueCalcul(
      resultat,
      erreur,
      (
        onValider: (marie, enfants, salaire) =>
            _calculer(marie, enfants, salaire),
        onListe: () => _afficherListe(),
        onFinSession: () => _finSession(),
      ),
    );
  }

  // Tax calculation performed entirely in the browser (layer
  // [métier]): no server calls, unlike the first version
  // of this client (HTTP client 1 / HTTP client 2)
  void _calculer(String marie, int enfants, int salaire) {
    // _metier must have been initialized: the [calcul] view
    // only after _initialiserMetier (so [!] is definitely here, as is [!]
    // from TypeScript to [this.métier!])
    final resultat = _metier!.calculerImpot(marie, enfants, salaire);
    // [différence avec TypeScript] Vues.ts builds the simulation with a
    // simple object spread: `{ married, children, salary, ...result }`.
    // Here, [Simulation] is a RECORD (simulation.dart), not a Map: a
    // record cannot be “populated” via spread from a Map, and
    // its fields are named SANS with accents (tax, surcharge, etc.), whereas
    // [resultat], returned by metier.dart, is a Map whose keys
    // GARDENT contain accents (“impôt,” “surcôte,” etc.; see metier.dart). We
    // thus reconstructs the record field by field—a bit more verbose
    // than a spread, but fully verified by the compiler (a
    // is a compilation error, not a surprise at
    // runtime).
    final Simulation simulation = (
      marie: marie,
      enfants: enfants,
      salaire: salaire,
      impot: resultat['impôt'] as num,
      surcote: resultat['surcôte'] as num,
      decote: resultat['décôte'] as num,
      reduction: resultat['réduction'] as num,
      taux: resultat['taux'] as num,
    );
    // stored in the browser (such as `simulations.add(...)` in
    // main3.dart)
    _simulations.add(simulation);
    _afficherCalcul(simulation);
  }

  // --------------------------------------------------------------------
  // view [liste des simulations]
  // --------------------------------------------------------------------
  // No more server calls here: the list is already entirely in
  // [_simulations]
  void _afficherListe([String? erreur]) {
    vues.vueListe(
      _simulations,
      erreur,
      (
        onSupprimer: (numero) => _supprimer(numero),
        onCalcul: () => _afficherCalcul(),
        onFinSession: () => _finSession(),
      ),
    );
  }

  void _supprimer(int numero) {
    // local deletion of [_simulations], as
    // `simulations.removeAt(number)` in main3.dart - still none
    // server call
    _simulations.removeAt(numero);
    _afficherListe();
  }

  // --------------------------------------------------------------------
  // end of session
  // --------------------------------------------------------------------
  void _finSession() {
    // end of session on the client side only: we ignore the [métier] layer
    // (i.e., tax data) and the simulations already calculated, then
    // return to the authentication screen. This client does not need to
    // to query the server for this.
    _metier = null;
    _simulations = [];
    _afficherAuthentification();
  }

  // --------------------------------------------------------------------
  // formats the field [réponse] from a server error response, which
  // is either a string or a list of strings (see PHP controllers)
  // --------------------------------------------------------------------
  String _formaterErreur(Object? reponse) {
    if (reponse is List) {
      return reponse.join(' ; ');
    }
    return '$reponse';
  }
}

    6.6. Conclusion

    The four clients in this chapter illustrate, with Dart, exactly the same progression as with TypeScript: from server-side processing to client-side processing, then from the CLI script to the full web client. The key differences to note are the manual tracking of the session cookie (necessary in CLI, unnecessary in a browser), the [.toJS] conversion enforced by [dart:js_interop] to bridge the Dart/JavaScript boundary in views, and the complete absence of polyfills during web compilation—a clear advantage of the Dart toolchain over the webpack/babel pipeline.
